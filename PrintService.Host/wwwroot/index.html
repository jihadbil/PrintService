⚛️ كود الاستدعاء في React 
// انسخ print-client.js إلى تطبيق React
import { createPrintClient } from "./print-client.js";
const client = createPrintClient({ systemId: "restaurant" });
client.saveSettings({
  printerName: "Microsoft Print to PDF",
  copies: 1,
  page: { paperName: "A4", margins: { leftMm: 5, rightMm: 5 } }
});
const job = client.createTextJob({
  jobId: "invoice:123", // أعد استخدام المهمة نفسها للمحاولات
  text: "فاتورة مبيعات\nالإجمالي: 150 د.ل"
});
const result = await client.print(job);
